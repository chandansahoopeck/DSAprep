# Redis-backed Rate Limiter (Sliding Window)
class SlidingWindowRateLimiter:
def __init__(self, redis_client, max_requests, window_seconds):
self.redis = redis_client
self.max_requests = max_requests
self.window = window_seconds

def is_allowed(self, user_id):
key = f"rate_limit:{user_id}"
now = time.time()
window_start = now - self.window

# Remove old entries
self.redis.zremrangebyscore(key, 0, window_start)

# Count requests in window
count = self.redis.zcard(key)

if count < self.max_requests:
self.redis.zadd(key, {str(now): now})
self.redis.expire(key, self.window)
return True
return False

LRU/LFU Cache with TTL

Phase 3's context/memory eviction layer

Weeks 3-4

LRU (Least Recently Used)

  • Doubly linked list + hash map
  • O(1) get/put operations
  • Evict least recently accessed

LFU (Least Frequently Used)

  • Frequency counter + min heap
  • O(log n) operations
  • Evict least frequently accessed

# LRU Cache Implementation
class LRUCache:
def __init__(self, capacity: int, ttl: int = 300):
self.capacity = capacity
self.ttl = ttl
self.cache = {} # key -> (value, timestamp, node)
self.size = 0

# Doubly linked list
self.head = Node() # Dummy head (MRU)
self.tail = Node() # Dummy tail (LRU)
self.head.next = self.tail
self.tail.prev = self.head

def get(self, key: int) -> int:
if key not in self.cache:
return -1

value, timestamp, node = self.cache[key]

# Check TTL
if time.time() - timestamp > self.ttl:
self._remove_node(node)
del self.cache[key]
self.size -= 1
return -1

# Move to head (most recently used)
self._move_to_head(node)
return value

def put(self, key: int, value: int) -> None:
if key in self.cache:
_, _, node = self.cache[key]
node.value = value
self._move_to_head(node)
self.cache[key] = (value, time.time(), node)
else:
if self.size >= self.capacity:
# Evict LRU
lru_node = self.tail.prev
self._remove_node(lru_node)
del self.cache[lru_node.key]
self.size -= 1

new_node = Node(key, value)
self._add_node(new_node)
self.cache[key] = (value, time.time(), new_node)
self.size += 1

# Priority Queue with Heap
import heapq
from dataclasses import dataclass, field
from typing import Any

@dataclass(order=True)
class PriorityTask:
priority: int
timestamp: float = field(compare=False)
task_id: str = field(compare=False)
payload: Any = field(compare=False)
retries: int = field(default=0, compare=False)

class JobScheduler:
def __init__(self):
self.queue = [] # Min-heap
self.task_map = {} # task_id -> task

def schedule(self, task_id: str, priority: int, payload: Any, delay: int = 0):
task = PriorityTask(
priority=priority,
timestamp=time.time() + delay,
task_id=task_id,
payload=payload
)
heapq.heappush(self.queue, task)
self.task_map[task_id] = task

def execute_next(self):
if not self.queue:
return None

task = heapq.heappop(self.queue)

# Check if ready to execute
if task.timestamp > time.time():
heapq.heappush(self.queue, task)
return None

del self.task_map[task.task_id]
return task

def retry_with_backoff(self, task: PriorityTask, max_retries: int = 3):
if task.retries >= max_retries:
# Move to dead letter queue
self._move_to_dlq(task)
return

task.retries += 1
delay = (2 ** task.retries) # Exponential backoff
task.timestamp = time.time() + delay
heapq.heappush(self.queue, task)

Consistent Hashing Router

Design a load balancer - routing across multiple vLLM replicas

Weeks 5-6

Use Cases:

  • Distributed cache clusters
  • Database sharding
  • Load balancing across replicas
  • Minimal redistribution on scaling

# Consistent Hashing Implementation
import hashlib
import bisect

class ConsistentHashRouter:
def __init__(self, nodes=None, replicas=150):
self.replicas = replicas
self.ring = []
self.node_map = {}

if nodes:
for node in nodes:
self.add_node(node)

def _hash(self, key):
return int(hashlib.md5(key.encode()).hexdigest(), 16)

def add_node(self, node):
for i in range(self.replicas):
virtual_key = f"{node}:{i}"
hash_key = self._hash(virtual_key)
bisect.insort(self.ring, hash_key)
self.node_map[hash_key] = node

def remove_node(self, node):
for i in range(self.replicas):
virtual_key = f"{node}:{i}"
hash_key = self._hash(virtual_key)
if hash_key in self.ring:
self.ring.remove(hash_key)
del self.node_map[hash_key]

def get_node(self, key):
if not self.ring:
return None

hash_key = self._hash(key)

# Find the first node >= hash_key
idx = bisect.bisect(self.ring, hash_key)

# Wrap around if necessary
if idx >= len(self.ring):
idx = 0

return self.node_map[self.ring[idx]]

def scale_out(self, new_nodes):
"""Add new nodes with minimal redistribution"""
for node in new_nodes:
self.add_node(node)
# Only keys that mapped to old positions need migration
